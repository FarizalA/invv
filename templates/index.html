Salin Kode Apps Script Lengkap (Sheets + Drive + Live Monitor)

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(15000);
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getActiveSheet();
    if (sheet.getLastRow() === 0) {
      var headers = ["Waktu Input", "No. Invoice / Ref", "Tanggal Invoice", "Nama Klien / Pembeli", "Alamat / Lokasi", "Deskripsi / Layanan", "Kategori", "Subtotal", "PPN / Pajak", "Diskon", "Total Pemasukan (Rp)", "Bank / Rekening Tujuan", "Status Pembayaran", "Link Bukti (Drive)", "Catatan"];
      sheet.appendRow(headers);
      sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#0284c7").setFontColor("#FFFFFF").setHorizontalAlignment("center");
      sheet.setFrozenRows(1);
    }
    var driveUrl = "-";
    if (data.file_base64 && data.file_base64.length > 50) {
      try {
        var folderName = "Bukti Pemasukan Invoice";
        var folders = DriveApp.getFoldersByName(folderName);
        var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
        var decoded = Utilities.base64Decode(data.file_base64);
        var safeFileName = (data.invoice_number ? data.invoice_number.replace(/[^a-zA-Z0-9_-]/g, "_") + "_" : "") + (data.file_name || "bukti.jpg");
        var blob = Utilities.newBlob(decoded, data.file_mime || "application/octet-stream", safeFileName);
        var driveFile = folder.createFile(blob);
        driveFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        driveUrl = driveFile.getUrl();
      } catch (err) { driveUrl = "Gagal upload: " + err.toString(); }
    }
    var nowStr = Utilities.formatDate(new Date(), "Asia/Jakarta", "yyyy-MM-dd HH:mm:ss");
    sheet.appendRow([nowStr, data.invoice_number || "-", data.invoice_date || "-", data.customer_name || "Pelanggan Umum", data.address || "-", data.description || "-", data.category || "Pemasukan", Number(data.subtotal) || 0, Number(data.tax) || 0, Number(data.discount) || 0, Number(data.total_amount) || 0, data.payment_method || "-", data.status || "Lunas", driveUrl, data.notes || "-"]);
    var lastRow = sheet.getLastRow();
    sheet.getRange(lastRow, 8, 1, 4).setNumberFormat("#,##0");
    if (driveUrl.indexOf("http") === 0) {
      sheet.getRange(lastRow, 14).setFormula('=HYPERLINK("' + driveUrl + '"; "Lihat Bukti Drive")');
    }
    return ContentService.createTextOutput(JSON.stringify({status: "success", row: lastRow, drive_url: driveUrl})).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getActiveSheet();
    var data = sheet.getDataRange().getValues();
    if (data.length <= 1) {
      return ContentService.createTextOutput(JSON.stringify({spreadsheet_url: ss.getUrl(), rows: []})).setMimeType(ContentService.MimeType.JSON);
    }
    var rows = [];
    for (var i = 1; i < data.length; i++) {
      var r = data[i];
      var rawDate = r[2];
      var dateStr = rawDate instanceof Date ? Utilities.formatDate(rawDate, "Asia/Jakarta", "yyyy-MM-dd") : String(rawDate);
      
      // Deteksi fleksibel jika kolom Alamat ada di index 4 atau format lama
      var hasAddressCol = data[0].length >= 15 || (data[0][4] && String(data[0][4]).toLowerCase().indexOf("alamat") !== -1);
      var cAddress = hasAddressCol ? String(r[4]) : "-";
      var offset = hasAddressCol ? 1 : 0;

      rows.push({
        row_index: i + 1,
        timestamp: String(r[0]),
        invoice_number: String(r[1]),
        invoice_date: dateStr,
        customer_name: String(r[3]),
        address: cAddress,
        description: String(r[4 + offset]),
        category: String(r[5 + offset]),
        subtotal: Number(r[6 + offset]) || 0,
        tax: Number(r[7 + offset]) || 0,
        discount: Number(r[8 + offset]) || 0,
        total_amount: Number(r[9 + offset]) || 0,
        payment_method: String(r[10 + offset]),
        status: String(r[11 + offset]),
        drive_url: String(r[12 + offset]),
        notes: String(r[13 + offset])
      });
    }
    rows.reverse();
    return ContentService.createTextOutput(JSON.stringify({spreadsheet_url: ss.getUrl(), rows: rows})).setMimeType(ContentService.MimeType.JSON);
  } catch(e) {
    return ContentService.createTextOutput(JSON.stringify({rows: []})).setMimeType(ContentService.MimeType.JSON);
  }
}
